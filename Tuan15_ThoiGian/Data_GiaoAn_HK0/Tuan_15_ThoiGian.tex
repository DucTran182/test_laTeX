% Đường bao vỏ đồng hồ (tròn / đa giác đều / ngôi sao): #1 kiểu vẽ, #2 bán kính
\newcommand\TTdh@duong[2]{%
  \ifx\TTdh@hinh\TTdh@s@circle
    \path[#1] (0,0) circle ({#2});%
  \else\ifx\TTdh@hinh\TTdh@s@star
    \pgfmathtruncatemacro\TTdh@n{2*\TTdh@canh-1}%
    \path[#1] plot[samples at={0,...,\TTdh@n}]
      ({\TTdh@xoay+\x*180/\TTdh@canh}:{(#2)*(1-\TTdh@lom*mod(\x,2))}) -- cycle;%
  \else
    \pgfmathtruncatemacro\TTdh@n{\TTdh@canh-1}%
    \path[#1] plot[samples at={0,...,\TTdh@n}]
      ({\TTdh@xoay+\x*360/\TTdh@canh}:{#2}) -- cycle;%
  \fi\fi}
% Một chiếc kim: #1 góc, #2 chiều dài, #3 bề rộng, #4 màu
\newcommand\TTdh@kimve[4]{%
  \ifx\TTdh@kim\TTdh@s@bar
    \draw[black, opacity=.18, line width=#3 cm, line cap=round, shift={(0.025,-0.035)}, rotate=#1] (0,0) -- (#2,0);%
    \draw[#4, line width=#3 cm, line cap=round, rotate=#1] (0,0) -- (#2,0);%
  \else\ifx\TTdh@kim\TTdh@s@leaf
    \fill[black, opacity=.15, shift={(0.025,-0.035)}, rotate=#1]
      (-0.1*#2,0) .. controls (0.1*#2,0.6*#3) and (0.3*#2,0.62*#3) .. (0.45*#2,0.5*#3)
      -- (#2,0) -- (0.45*#2,-0.5*#3) .. controls (0.3*#2,-0.62*#3) and (0.1*#2,-0.6*#3) .. cycle;%
    \fill[#4, rotate=#1]
      (-0.1*#2,0) .. controls (0.1*#2,0.6*#3) and (0.3*#2,0.62*#3) .. (0.45*#2,0.5*#3)
      -- (#2,0) -- (0.45*#2,-0.5*#3) .. controls (0.3*#2,-0.62*#3) and (0.1*#2,-0.6*#3) .. cycle;%
    \draw[white, opacity=.45, line width=.35pt, rotate=#1] (0.05*#2,0) -- (0.85*#2,0);%
  \else\ifx\TTdh@kim\TTdh@s@arrow
    \draw[#4, line width=#3 cm, rotate=#1, -{Stealth[length=2.3mm,width=2mm]}] (0,0) -- (#2,0);%
  \else
    \fill[black, opacity=.18, shift={(0.03,-0.04)}, rotate=#1, rounded corners=.5pt]
      (-0.18*#2,-0.5*#3) -- (#2-1.4*#3,-0.34*#3) -- (#2,0) -- (#2-1.4*#3,0.34*#3) -- (-0.18*#2,0.5*#3) -- cycle;%
    \fill[#4, rotate=#1, rounded corners=.5pt]
      (-0.18*#2,-0.5*#3) -- (#2-1.4*#3,-0.34*#3) -- (#2,0) -- (#2-1.4*#3,0.34*#3) -- (-0.18*#2,0.5*#3) -- cycle;%
  \fi\fi\fi}
\newcommand\TTdongho[3][]{%
  \begin{scope}
    \pgfkeys{/TTdh/.cd,#1}%
    \pgfmathsetmacro\TTdh@Rf{\TTdh@R*(1-\TTdh@day)}%
    \pgfmathsetmacro\TTdh@ag{90-30*(#2)-0.5*(#3)}%
    \pgfmathsetmacro\TTdh@ap{90-6*(#3)}%
    % bóng đổ, vỏ (đổ màu chuyển sáng - tối như kim loại), mặt số
    \TTdh@duong{fill=black, opacity=\TTdh@bong, shift={(0.05*\TTdh@R,-0.07*\TTdh@R)}}{\TTdh@R}%
    \TTdh@duong{top color=\TTdh@vien!62!white, bottom color=\TTdh@vien!88!black,
                draw=\TTdh@vien!65!black, line width=.4pt, line join=round}{\TTdh@R}%
    \TTdh@duong{inner color=white, outer color=\TTdh@mat, draw=\TTdh@vien!55!black,
                line width=.45pt, line join=round}{\TTdh@Rf}%
    % vạch chia
    \ifx\TTdh@vach\TTdh@s@ticks
      \foreach \i in {0,...,59}
        \draw[\TTdh@mauvach, line width=.3pt] ({\i*6}:{0.91*\TTdh@Rf}) -- ({\i*6}:{0.965*\TTdh@Rf});%
      \foreach \i in {0,...,11}
        \draw[\TTdh@mauvach, line width=.9pt] ({\i*30}:{0.865*\TTdh@Rf}) -- ({\i*30}:{0.965*\TTdh@Rf});%
      \foreach \i in {0,...,3}
        \draw[\TTdh@mauvach, line width=1.4pt] ({\i*90}:{0.84*\TTdh@Rf}) -- ({\i*90}:{0.965*\TTdh@Rf});%
    \else\ifx\TTdh@vach\TTdh@s@hour
      \foreach \i in {0,...,11}
        \draw[\TTdh@mauvach, line width=1pt, line cap=round] ({\i*30}:{0.84*\TTdh@Rf}) -- ({\i*30}:{0.95*\TTdh@Rf});%
    \else\ifx\TTdh@vach\TTdh@s@dots
      \foreach \i in {1,2,4,5,7,8,10,11}
        \fill[\TTdh@mauvach] ({90-\i*30}:{\TTdh@vs*\TTdh@Rf}) circle ({0.04*\TTdh@R});%
    \fi\fi\fi
    % số
    \ifx\TTdh@so\TTdh@s@all
      \foreach \i in {1,...,12}
        \node[font=\TTdh@chu, text=\TTdh@mauso, inner sep=0pt] at ({90-\i*30}:{\TTdh@vs*\TTdh@Rf}) {\i};%
    \else\ifx\TTdh@so\TTdh@s@four
      \foreach \i in {3,6,9,12}
        \node[font=\TTdh@chu, text=\TTdh@mauso, inner sep=0pt] at ({90-\i*30}:{\TTdh@vs*\TTdh@Rf}) {\i};%
    \fi\fi
    % kim (kim phút dừng trước vòng số nên không đè lên số)
    \ifx\TTdh@kim\TTdh@s@none
      \fill[\TTdh@tam] (0,0) circle ({0.035*\TTdh@R});%
    \else
      \pgfmathsetmacro\TTdh@Lp{\TTdh@vs*\TTdh@Rf-\TTdh@hs}%
      \pgfmathsetmacro\TTdh@Lg{0.64*\TTdh@Lp}%
      \ifx\TTdh@kim\TTdh@s@leaf
        \pgfmathsetmacro\TTdh@Wg{0.13*\TTdh@R}\pgfmathsetmacro\TTdh@Wp{0.095*\TTdh@R}%
      \else\ifx\TTdh@kim\TTdh@s@arrow
        \pgfmathsetmacro\TTdh@Wg{0.042*\TTdh@R}\pgfmathsetmacro\TTdh@Wp{0.03*\TTdh@R}%
      \else\ifx\TTdh@kim\TTdh@s@bar
        \pgfmathsetmacro\TTdh@Wg{0.08*\TTdh@R}\pgfmathsetmacro\TTdh@Wp{0.055*\TTdh@R}%
      \else
        \pgfmathsetmacro\TTdh@Wg{0.1*\TTdh@R}\pgfmathsetmacro\TTdh@Wp{0.07*\TTdh@R}%
      \fi\fi\fi
      \TTdh@kimve{\TTdh@ag}{\TTdh@Lg}{\TTdh@Wg}{\TTdh@kg}%
      \TTdh@kimve{\TTdh@ap}{\TTdh@Lp}{\TTdh@Wp}{\TTdh@kp}%
      \fill[\TTdh@tam] (0,0) circle ({0.065*\TTdh@R});%
      \fill[white!70!gray] (0,0) circle ({0.022*\TTdh@R});%
    \fi
  \end{scope}}
