\subsection{MỤC TIÊU CẦN ĐẠT}
\begin{itemize}
    \item \textbf{Kiến thức:} HS ôn tập phép cộng, phép trừ trong phạm vi 5 (qua tranh và qua số).
    \item \textbf{Kiến thức:} HS nhận biết và viết đúng dấu cộng $(+)$, dấu trừ $(-)$, dấu bằng $(=)$; lập được phép cộng, phép trừ phù hợp với tranh hoặc tình huống.
    \item \textbf{Năng lực:} HS nói được một câu trọn vẹn khi đố bạn và khi giải thích cách làm; biết hợp tác, thay phiên khi chơi.
    \item \textbf{Phẩm chất:} HS hào hứng, tự tin, trung thực khi báo điểm, biết cổ vũ bạn.
\end{itemize}

\subsection{TIẾN TRÌNH DẠY HỌC}
\textbf{Đồ dùng dạy học:} bút, khăn ẩm, phiếu bài tập, giáo án, con dấu, hộp màu, bảng con, thẻ số từ 0 đến 5, thẻ dấu $+$, $-$, $=$ (có băng dính hoặc nam châm), kẹo hoặc đồ vật thật để minh họa, laptop/TV, loa.

\begin{luuy}
    \textbf{Cách tính điểm cả buổi:} lớp chia thành 2 nhóm, mỗi nhóm có một “cây ngôi sao” trên bảng. HS trả lời đúng, làm nhanh, hợp tác tốt đều được dán sao. Cuối buổi nhóm nhiều sao nhất được khen thưởng.
\end{luuy}
